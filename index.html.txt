    <script>
        (function() {
            // ВЕЧНЫЙ СЧЕТЧИК
            let total = 0;
            let throws = 0;
            
            try {
                let saved = localStorage.getItem('kakaha_total');
                if (saved) total = parseInt(saved) || 0;
            } catch(e) {}
            
            // ЭЛЕМЕНТЫ
            const counter = document.getElementById('counter');
            const throwsStat = document.getElementById('throwsStat');
            const hitsStat = document.getElementById('hitsStat');
            const totalStat = document.getElementById('totalStat');
            const photoBox = document.getElementById('photoBox');
            const effects = document.getElementById('effects');
            const clicker = document.getElementById('clicker');
            
            // ОБНОВЛЕНИЕ
            function updateDisplay() {
                counter.textContent = total;
                throwsStat.textContent = throws;
                hitsStat.textContent = total;
                totalStat.textContent = total;
                
                try {
                    localStorage.setItem('kakaha_total', total.toString());
                } catch(e) {}
            }
            
            // ЗВУК
            function playSound() {
                try {
                    let audio = new (window.AudioContext || window.webkitAudioContext)();
                    let osc = audio.createOscillator();
                    let gain = audio.createGain();
                    
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(150, audio.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(70, audio.currentTime + 0.2);
                    
                    gain.gain.setValueAtTime(0.2, audio.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audio.currentTime + 0.2);
                    
                    osc.connect(gain);
                    gain.connect(audio.destination);
                    
                    osc.start();
                    osc.stop(audio.currentTime + 0.2);
                    
                    if (audio.state === 'suspended') audio.resume();
                } catch(e) {}
            }
            
            // ПОПАДАНИЕ
            function addHit() {
                let box = photoBox.getBoundingClientRect();
                if (box.width === 0) return;
                
                // 2-3 попадания за раз
                let count = 2 + Math.floor(Math.random() * 2);
                
                for (let i = 0; i < count; i++) {
                    setTimeout(() => {
                        let x = Math.random() * box.width;
                        let y = Math.random() * box.height;
                        
                        // ПЯТНО
                        let stain = document.createElement('div');
                        stain.className = 'stain';
                        stain.style.left = (x - 20) + 'px';
                        stain.style.top = (y - 20) + 'px';
                        
                        // КАКАХА
                        let emoji = document.createElement('div');
                        emoji.className = 'poop';
                        emoji.style.left = (x - 17) + 'px';
                        emoji.style.top = (y - 25) + 'px';
                        emoji.textContent = '💩';
                        
                        effects.appendChild(stain);
                        effects.appendChild(emoji);
                        
                        setTimeout(() => {
                            stain.remove();
                            emoji.remove();
                        }, 800);
                    }, i * 50);
                }
                
                photoBox.classList.add('shake');
                setTimeout(() => {
                    photoBox.classList.remove('shake');
                }, 200);
            }
            
            // КЛИК
            function handleClick(e) {
                e.preventDefault();
                
                total++;
                throws++;
                
                playSound();
                addHit();
                updateDisplay();
                
                try { window.navigator.vibrate(20); } catch(e) {}
            }
            
            // СОБЫТИЯ
            clicker.addEventListener('click', handleClick);
            clicker.addEventListener('touchstart', (e) => {
                e.preventDefault();
                handleClick(e);
            });
            
            // СТАРТ
            updateDisplay();
        })();
    </script>
